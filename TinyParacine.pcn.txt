# Paracine Stage-1 compiler driver for Windows PowerShell 5.1+.
# The compiler frontend is entirely written in Paracine. This script supplies
# source-file I/O because the current Stage-0 Paracine compiler has no stable
# Paracine std.fs/read_file API yet. Stage-0 remains the native PE backend.

param(
    [string]Stage0 = "Paracine",
    [Parameter(Mandatory=true)][string]Source,
    [string]Output = "generated.exe",
    [switch]Run
)

ErrorActionPreference = 'Stop'

function Require-Success([string]label) {
    if (LASTEXITCODE -ne 0) {
        throw "label failed: exit code LASTEXITCODE"
    }
}

command = Get-Command Stage0 -ErrorAction Stop
compiler = command.Source

inputFile = (Resolve-Path -LiteralPath Source -ErrorAction Stop).Path
outputFile = [IO.Path]::GetFullPath(Output)

outputFolder = Split-Path -Parent outputFile
[IO.Directory]::CreateDirectory(outputFolder) | Out-Null

workspace = Join-Path outputFolder '.tinyparacine-build'
stage1Folder = Join-Path workspace 'stage1'

[IO.Directory]::CreateDirectory(stage1Folder) | Out-Null

# Normalize line endings and ensure a final statement terminator.
inputText = [IO.File]::ReadAllText(inputFile)

inputText = inputText
    .Replace("`r`n", "`n")
    .Replace("`r", "`n")

if (!inputText.EndsWith("`n")) {
    inputText += "`n"
}

# Encode file content as a Paracine text literal.
escaped = inputText.Replace('\', '\\').Replace('"', '\"')
escaped = escaped.Replace("`t", '\t').Replace("`n", '\n')

templateFile = Join-Path PSScriptRoot 'stage1\tiny_compiler.pcn'
template = [IO.File]::ReadAllText(templateFile)

lines = template -split "`n"

replaced = 0

for (i = 0; i -lt lines.Length; i++) {
    if (lines[i] -match '^    let input: text = ') {
        lines[i] = '    let input: text = "' + escaped + '"'
        replaced++
    }
}

if (replaced -ne 1) {
    throw 'Stage-1 template input location missing or ambiguous'
}

stage1File = Join-Path stage1Folder 'tiny_compiler.pcn'

utf8 = New-Object System.Text.UTF8Encoding(false)

[IO.File]::WriteAllText(
    stage1File,
    (lines -join "`n"),
    utf8
)

stage1Executable = Join-Path stage1Folder 'tiny_compiler.exe'
generatedPcn = Join-Path workspace 'generated.pcn'
compilerError = Join-Path workspace 'compiler-errors.txt'

Write-Host '[1/4] Compile the Stage-1 Paracine-written compiler...'

& compiler pe stage1File -O0 -o stage1Executable

Require-Success 'Stage-1 native build'

Write-Host '[2/4] Run Stage-1 to compile the source file...'

proc = Start-Process `
    -FilePath stage1Executable `
    -Wait `
    -NoNewWindow `
    -PassThru `
    -RedirectStandardOutput generatedPcn `
    -RedirectStandardError compilerError

if (proc.ExitCode -ne 0) {
    diagnostic = [IO.File]::ReadAllText(compilerError)
    throw "Stage-1 compile failed ((proc.ExitCode)):`ndiagnostic"
}

generated = [IO.File]::ReadAllText(generatedPcn)

if (!generated.StartsWith("module generated`n")) {
    throw 'Unexpected Stage-1 output; generated source was not produced'
}

Write-Host '[3/4] Check generated Paracine source...'

& compiler check generatedPcn

Require-Success 'Generated source check'

Write-Host '[4/4] Compile generated Paracine into native PE32+...'

& compiler pe generatedPcn -O3 -o outputFile

Require-Success 'Generated native executable'

Write-Host "Generated .pcn: generatedPcn"
Write-Host "Generated .exe: outputFile"

if (Run) {
    Write-Host 'Running generated executable...'

    & outputFile

    Write-Host "Program exit code: LASTEXITCODE"
}
